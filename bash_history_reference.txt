import pandas as pd
import matplotlib.pyplot as plt
# Load data
df = pd.read_csv("airtravel.csv")
# Clean column names
df.columns = df.columns.str.replace('"', '').str.strip()
# Convert to numeric
for col in ["1958", "1959", "1960"]:;     df[col] = pd.to_numeric(df[col], errors="coerce")
# Add total passengers
df["total_passengers"] = df["1958"] + df["1959"] + df["1960"]
# Insights
busiest_month = df.loc[df["total_passengers"].idxmax(), "Month"]
print("Busiest Month:", busiest_month)
print("\nYearly Averages:")
print(df[["1958", "1959", "1960"]].mean())
# Save cleaned data
df.to_csv("airtravel_cleaned.csv", index=False)
# Save summary
df.describe().to_csv("airtravel_summary.csv")
# Visualization
plt.figure(figsize=(10,5))
plt.plot(df["Month"], df["1958"], marker="o", label="1958")
plt.plot(df["Month"], df["1959"], marker="o", label="1959")
plt.plot(df["Month"], df["1960"], marker="o", label="1960")
plt.title("Airline Passenger Trend (1958ΓÇô1960)")
plt.xlabel("Month")
plt.ylabel("Passengers")
plt.legend() plt.xticks(rotation=45)
plt.tight_layout() plt.savefig("airtravel_chart.png")
